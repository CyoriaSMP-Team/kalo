Furniture

Basic Chair

ruby_chair:
  type: furniture
  display:
    name: "<gold>Ruby Chair</gold>"
  model:
    custom: "furniture/ruby_chair"
  furniture:
    rotatable: true
    restricted_rotation: strict
    seat:
      height: 0.5
      offset: [0.0, 0.5, 0.0]
    hitbox:
      barriers: [[0,0,0]]
  java:
    mode: native

Storage Furniture

ruby_chest:
  type: furniture
  display:
    name: "<gold>Ruby Chest</gold>"
  model:
    custom: "furniture/ruby_chest"
  furniture:
    storage:
      type: STORAGE
      rows: 5
      title: "<red>Storage</red>"
  java:
    mode: native
